cmake_minimum_required(VERSION 3.16)
project(go2_fusion_cpp LANGUAGES CXX)
if(POLICY CMP0104)
  cmake_policy(SET CMP0104 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GO2_FUSION_ROS "Build ROS 2 adapter" OFF)
option(GO2_FUSION_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(GO2_FUSION_CUDA "Build optional, benchmark-gated visibility accelerator" OFF)
add_library(go2_fusion_core src/core.cpp)
target_include_directories(go2_fusion_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(go2_fusion_core PRIVATE -Wall -Wextra -Wpedantic)
if(GO2_FUSION_SANITIZE)
  target_compile_options(go2_fusion_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(go2_fusion_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(fusion_benchmark tools/benchmark.cpp)
target_link_libraries(fusion_benchmark PRIVATE go2_fusion_core)
if(GO2_FUSION_CUDA)
  if(NOT CMAKE_CUDA_ARCHITECTURES)
    # Conservative Ampere default with PTX forward compatibility. Target builds
    # should select their measured device architecture explicitly.
    set(CMAKE_CUDA_ARCHITECTURES 86)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(go2_fusion_cuda src/cuda_visibility.cu)
  target_include_directories(go2_fusion_cuda PUBLIC include)
  target_compile_features(go2_fusion_cuda PUBLIC cxx_std_17)
  set_target_properties(go2_fusion_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(go2_fusion_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  target_link_libraries(go2_fusion_cuda PUBLIC CUDA::cudart)
  target_compile_definitions(go2_fusion_cuda PUBLIC GO2_FUSION_HAS_CUDA=1)
  target_link_libraries(fusion_benchmark PRIVATE go2_fusion_cuda)
  add_executable(cuda_visibility_benchmark tools/cuda_visibility_benchmark.cpp)
  target_link_libraries(cuda_visibility_benchmark PRIVATE go2_fusion_core go2_fusion_cuda)
  install(TARGETS go2_fusion_cuda cuda_visibility_benchmark ARCHIVE DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(fusion_core_test test/core_test.cpp)
  target_link_libraries(fusion_core_test PRIVATE go2_fusion_core)
  add_test(NAME fusion_core_test COMMAND fusion_core_test)
  if(GO2_FUSION_CUDA)
    add_executable(cuda_visibility_test test/cuda_visibility_test.cpp)
    target_link_libraries(cuda_visibility_test PRIVATE go2_fusion_cuda)
    add_test(NAME cuda_visibility_test COMMAND cuda_visibility_test)
  endif()
endif()
install(TARGETS go2_fusion_core fusion_benchmark EXPORT go2_fusion_coreTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(EXPORT go2_fusion_coreTargets NAMESPACE go2_fusion:: DESTINATION lib/cmake/go2_fusion_cpp)
if(GO2_FUSION_ROS)
  include(cmake/ros.cmake)
endif()
